na51089 bootrom -- function ledger
===================================

NT98560/561/562/566 (NA51089) mask ROM, 32 KiB, linked at 0x00000000 (home
alias 0x80000000).  All code is ARM (A32); there are no Thumb functions (see
bootrom.txt).  0x0000..0x031f is hand-written assembly (start.S); everything
from 0x0320 is gcc output with frame pointers.

Format: func_NN  <addr>  name(signature)  notes  [callers]
func_NN is the index in address order.  Callers are the containing functions
of every `bl`/tail `b` that reaches the entry ("dispatch" = via the 0x1c0
jump table).  Addresses are the first instruction.  (inferred) marks
behaviour read from the code but not confirmed on hardware or by a second
source.  Letters printed on UART0 are quoted 'X'.

Global names used below are defined in sram.txt (g_boot_source, g_flash,
g_sd_*, g_usb_*, g_bot_state, g_sense, ...).  MMIO block names (TOP, CG,
PAD, TIMER, WDT, GPIO, OTP, SPI, SDIO1, SDIO3, ETH, USB, ...) are defined in
regmap.txt.


Startup and utilities (start.S + lib)
-------------------------------------

func_000  0x003c  exc_hang()
          six one-instruction `b .` stubs at 0x3c/0x40/0x44/0x48/0x4c/0x50 for undef/swi/pabt/dabt/irq/fiq; ROM never enables IRQ/FIQ
          [callers: vectors]

func_001  0x0054  reset_handler()
          [asm] SVC + I/F masked; SCTLR &= ~(I|C|A|M); CG+0x74 |= bit18, TIMER+0x30 |= bit16, TIMER+0x100 = 3; *(0xF0060018) = 1; sp = 0xF07D3000; CG+0x70 |= bit16, CG+0x18 &= ~bit26; uart0_init_banner ("N"); optional flash reset pulse on GPIO9 (BS=0, OTP+0x50 bit9) or GPIO10 (BS=8/9): dir out, low, [100 ms if WDT+0x80[15:8] != 0] + 1 ms, high, 1 ms, input; falls into boot_dispatch
          [callers: reset vector]

func_021  0x14fc  popcount(u32 x)
          Kernighan bit count; flash_boot uses it on the NAND spare/bad-block marker word (> 6 ones = good block)
          [callers: flash_boot]

func_025  0x16e8  memcpy32(void *dst, const void *src, u32 n)
          word copy, n rounded down to 4; also exported at 0x7FE8.
          [callers: flash_boot, secure_boot_verify, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot, usb_get_descriptor, scsi_request_sense, scsi_inquiry, scsi_mode_sense6]

func_026  0x1714  memset32(void *dst, u8 c, u32 n)
          word fill with c replicated, n rounded down to 4.
          [callers: secure_boot_verify, eth_xmit, eth_boot, sha256]

func_038  0x277c  timer_start()
          save CG+0x74 -> 0xF07D4010, CG+0x74 |= bit18; save TIMER+0x30 -> 0xF07D4014, |= bit16; TIMER+0x110 = 1 (counter restarts from 0). All timeouts are "elapsed since timer_start"
          [callers: flash_boot, rsa_run, udelay, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot]

func_039  0x27c4  u32 timer_now()
          return TIMER+0x118 (free-running up-counter, 1 MHz assumed: 0xF423F = 1 s, 0x3D08FF = 4 s, 0x4C4B40 = 5 s, 0x1C9C380 = 30 s)
          [callers: flash_boot, rsa_run, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot]

func_040  0x27dc  timer_stop()
          TIMER+0x110 = 0, wait until it reads 0; restore TIMER+0x30 and CG+0x74 from 0xF07D4014/0xF07D4010
          [callers: flash_boot, rsa_run, udelay, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot, nvt_vendor_cdb]

func_041  0x2824  udelay(u32 us)
          timer_start; spin while timer_now() < us; timer_stop.
          [callers: reset_handler, flash_boot, hdr_script_run, wdt_arm_or_skip, eth_boot, emmc_boot_read, sd_read_blocks, sd_boot, emmc_boot, usb_boot]

func_042  0x284c  u8 rd8(const void *p)
          byte load done as aligned word load + shift (unaligned-safe on device memory)
          [callers: rd32le, rd16le]

func_043  0x2870  u32 rd32le(const void *p)
          4 x rd8, little-endian; eth_boot follows it with `rev` for network order
          [callers: eth_boot]

func_044  0x28b8  u16 rd16le(const void *p)
          2 x rd8
          [callers: eth_boot]

func_045  0x28e0  wr16le(void *p, u16 v)
          two read-modify-write byte stores
          [callers: eth_boot]

func_046  0x292c  wr8(void *p, u8 v)
          aligned-word read-modify-write byte store
          [callers: eth_boot]

func_047  0x295c  memcpy8(void *dst, const void *src, u32 n)
          unaligned memcpy built from word loads/stores and shifts (used for all packet field copies)
          [callers: eth_boot]

func_050  0x2b14  wr32le(void *p, u32 v)
          unaligned word store (byte RMW when p&3)
          [callers: eth_boot]

func_052  0x2c84  u32 swap16x2(u32 x)
          swaps the bytes inside each halfword (htons on a value from rd16le)
          [callers: eth_boot]


Boot dispatch and fallback
--------------------------

func_002  0x01a4  boot_dispatch()
          [asm label inside reset_handler] g_boot_source(0xF07D4004) = TOP+0[3:0]; `ldr pc,[0x1c0 + 4*BS]` (jump, lr not set); table at 0x1c0
          [callers: reset_handler (fall-through)]

func_005  0x02dc  hang_reserved_strap()
          [asm] `b .` for BS = 7, 0xD, 0xE, 0xF (prints nothing beyond "N"); 0x2d8/0x2e0 are nop padding, 0x2e4..0x31c is the start.S literal pool
          [callers: dispatch]

func_027  0x1748  prejump_restore()
          called by run_loader. Returns at once for g_boot_source 8/9 (eMMC); otherwise puts clocks/pads back to defaults: CG+0x20=0x10000, CG+0x24=0xf00, CG+0x3c=0x01f001f0, CG+0x40=0x1303f1f0, pulse CG+0x84 bits 0,2,14,29 (resets), CG+0x74 &= ~(bit0|bit2|bit14) (SPI, SDIO1, SDIO3 clocks off), PAD+0x68=0x01000001, PAD+0x40=0x10000, PAD+0x98=0x100000
          [callers: run_loader]

func_029  0x180c  next_boot_source()
          CG+0x74 |= bit17; WDT+0 = 0x5A960F00 (watchdog off); m = 1 << g_boot_source: m & 0x174D (BS 0,2,3,6,8,9,10,12) -> g=1 sd_boot; m & 0x820 (BS 5,11) -> g=0xC uart_boot; m & 0x10 (BS 4) -> g=5 usb_boot; else (BS 1, 7, 13+) -> g=4 eth_boot. Chain: flash/eMMC -> SD -> ETH -> USB(HS) -> UART -> SD ...
          [callers: boot_fail, wdt_arm_or_skip]

func_030  0x18b4  boot_fail(int code)
          UART0 THR = 'K', then '0'+code (no FIFO wait) -> "K2".."K9", code 10 prints "K:"; tail: next_boot_source() (never returns). Codes: 2 SD init, 3 MBR, 4 PBR, 5 first sector/boot read, 6 bad flash strap, 7 loader read / NAND no good copy, 8 ETH DMA/PHY or header-extension read, 9 checksum/tag, 10 (":") timeout
          [callers: run_loader, flash_boot, hdr_script_run, rsa_run, secure_boot_verify, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot]

func_032  0x20b4  wdt_arm_or_skip()
          called first by flash_boot and emmc_boot. TOP+0 bit12 (BST[12]) set: print 'W'. Clear: CG+0x74 |= bit17; if WDT+0x80[15:8] != 0 (a watchdog reset happened) print 'J' and next_boot_source() (skip internal storage); else WDT+0 = 0x5A961042 then |= 0x5A960001 (arm). Both: if OTP+0x50 bit8, udelay(10000)
          [callers: flash_boot, emmc_boot]


Loader check, header script
---------------------------

func_003  0x0200  check_and_run_loader(int fatal, int hwsum_valid, u32 hwsum)
          [asm, no fp frame] load = *(0xF0800204); secure_boot_verify(load)==0 -> run_loader; hwsum_valid: (hwsum&0xffff)!=0 -> fail; else sum(h[i]+i) over *(0xF0800224)/2 halfwords at load must be 0 and *(u16*)(load+0x30)==0xAA55; fail: fatal ? boot_fail(9) : return.
          [callers: flash_boot, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot]

func_004  0x02b0  run_loader()
          [asm tail of check_and_run_loader, also reached by bl] prejump_restore(); UART0 THR = 'T'; pc = *(u32*)load (header word 0). r4 = load address on entry
          [callers: check_and_run_loader (b/bl), check_and_run_loader]

func_028  0x17f0  int hdr_tag_ok(const u32 *script)
          script = header+0x38: return *(u16*)(script-8) == 0xAA55 ? 0 : -1 (checks header+0x30)
          [callers: flash_boot, sd_boot]

func_031  0x18e0  hdr_script_run(const u32 *script)
          header register script (DRAM init) run before the body is loaded. header+0x30 != 0xAA55 -> boot_fail(9). count = script[0][11:0]; entries from header+0x40, op = w0[27:24], addr = w0 & ~0x0F000000: 0 poll *addr until masked bits (w1: n=[23:20], bit i at (w1>>5i)&31, value (w1>>(24+i))&1); 1 *addr = w1; 2 udelay(w0[23:0]); 3 write bytes of w1 to addr+0/4/8/0xc; 4 issue cmd to reg at header+0x34 and poll; 5 write lane bytes of w1 to header+0x3c base regs; 6 same at 0xC0021000 (DDR PHY); 7 DQS/gate training via 0xC000130C/0xC0001310, only if script[0] bit19, log at header+0x0c (0x2..../0x3.... words, 0x70654321 end); 8 trims from efuse_read(w0&31); 9 apply trims to lane regs (clamp 0..31); 10 flag = w0 bit0. CG+0 &= ~bit3 at entry. (op semantics 4/7/9 partly inferred)
          [callers: flash_boot, sd_boot, emmc_boot]


Secure boot, crypto, efuse
--------------------------

func_018  0x1364  aes_load_efuse_key()
          for k in 0..3: efuse_key_select(1, k); efuse_read(0x10+k) -- efuse words 16..19 pushed into the AES key (inferred hidden key path)
          [callers: secure_boot_verify]

func_019  0x13c0  aes_cbc_decrypt(const u8 *src, u8 *dst, int len)
          AES engine 0xF0620000: per 16 B block +0 = 1 (reset, wait bit0 clear), +0 = 0x122 (decrypt mode), data -> +0x30..+0x3c, +4 = 1, wait +0xc bit0 (W1C); out +0x40..+0x4c XOR IV (0xF07D0080) -> dst; IV = ciphertext. Returns 0
          [callers: secure_boot_verify]

func_020  0x14b8  reverse256(const u8 *src, u8 *dst)
          dst[255-i] = src[i] for a 256-byte block (big-endian word reads, words stored backwards); RSA signature in/out
          [callers: secure_boot_verify]

func_033  0x2158  rsa_load_n(const u32 *src, int nwords)
          RSA 0xF06A0000: +0x14 = 0 (index), nwords to +0x10 then zero-pad to 64 words
          [callers: secure_boot_verify]

func_034  0x21d4  rsa_load_e(const u32 *src, int nwords)
          +0x1c = 0, words to +0x18, zero-pad to 64
          [callers: secure_boot_verify]

func_035  0x2250  rsa_run(const u32 *msg, int nwords)
          +0x24 = 0, words to +0x20, pad to 64; +4 = 1 (start); wait +0xc bit1 then bit0, each with a 10 s / 1 s timeout -> boot_fail(10)
          [callers: secure_boot_verify]

func_036  0x2338  rsa_read(u32 *dst, int nwords)
          +0x24 = 0, read nwords from +0x20 (via a 64-entry pointer array on the stack)
          [callers: secure_boot_verify]

func_037  0x239c  int secure_boot_verify(u8 *img)
          returns 1 if OTP+0x50 bit5 clear (secure boot off), else prints 'X' and verifies, returning 0 or ending in boot_fail(9). bit7: 'S' AES-128-CBC decrypt img+0x300..+0x8000 in place (IV = 32 B at img+0x1e0 (RSA) or +0x2c0 (AES), key from efuse 0x10..0x13). bit1: 'R' RSA-2048: N = img+0xc0, E = img+0x1c0, sig = img+0x1e0 (byte-reversed); fields zeroed, SHA-256(img, 0x8000) -> 0xF07D0060; bit3: 'C' SHA-256(N) must equal efuse words 0x14..0x1b; RSA engine result tail (0xF07D05e0) must equal the digest; if *(0xF07C0088) bit29, img+0x2e0 (zeroed) must equal a rehash. bit1 clear: 'A' AES: 32 B at img+0x2c0 decrypted with efuse key (IV 0) must equal the SHA-256. Fields restored afterwards.
          [callers: check_and_run_loader]

func_064  0x4df4  u32 otp_boot_cfg()
          return OTP+0x50: bit1 RSA, bit3 key-hash check, bit4 SPI clock from pad ('D'/'I'), bit5 secure boot, bit6 skip NOR 'p' step, bit7 encrypted area, bit8 10 ms delay, bit9 NOR reset pulse, bit11 SD 'F', bit14 eMMC 'R', bits[20:18] timeout multiplier
          [callers: flash_boot, wdt_arm_or_skip, secure_boot_verify, eth_boot, sd_boot, emmc_boot, usb_boot, uart_boot]

func_065  0x4e0c  efuse_key_select(u32 a, u32 b)
          OTP+0x40 = a, OTP+0x44 = b (used only by aes_load_efuse_key)
          [callers: aes_load_efuse_key]

func_116  0x6d68  sha256(const void *data, int len, u32 out[8])
          HASH 0xF0670000, CG+0x78 |= bit27; +0 = 0x100012; 16 words per block to +0x90, wait +0xc bit0 (W1C); software padding (0x80, bit length, `rev`); digest +0x70..+0x8c -> out; clock off. Exported at 0x7FBC
          [callers: secure_boot_verify]

func_117  0x6eb4  u32 efuse_read(int idx)
          OTP 0xF0660000: +0x20 = 1, wait +0x30 bit0; +4 = idx&31; +0x20 |= 0x10; wait +0x30 bit1; v = +8; +0x20 = 0; wait +0x30 bit0 clear. Exported at 0x7FE4..0x1b), aes_load_efuse_key
          [callers: aes_load_efuse_key, hdr_script_run, secure_boot_verify]

func_069  0x4ebc  int uart0_rx_ready()

          [callers: uart_boot]

func_070  0x4ed8  uart0_reinit()

UART download (BS 0xC)
----------------------

func_071  0x4f28  uart_boot()
          BS 0xC or fallback. uart0_reinit; clear ctx 0xF07D4170..+0x20; timeout = (OTP[20:18]+1)*30 s; bytes -> LE words -> 16-byte CDB ring (4 words); every word is also a CDB word, even payload words (no resync). CDB sum = 8 halfwords + 0x1c, 0 = good. 0x81c0 (whole w0) IS_NVT: echo with fixed sum, then TOP+0xF0 (chip id) or 0; 0x01c0 OPEN_DEVICE: timer_stop (no more timeout), echo; lo16 0x83c0 ADDR_READ: echo, then addr(4) + *(u32*)addr(4) -- read happens even on a bad sum; 0x03c0 ADDR_WRITE: echo, *addr = val (also on a bad sum); 0x02c0 WRLOADER: addr = w0>>16 | w1<<16, words = (w1>>16 | w2<<16)/4, following words stored at addr, '.' on UART1 every 256 words, saved CDB echoed when complete; 0x82c0 WAIT_DONE: echo, leave. Bad sum: 4 zero words instead of the echo. Then memcpy32(0xF0800200, addr, 0x28); check_and_run_loader(1, 0, 0). Timeout -> boot_fail(10)
          [callers: dispatch, next_boot_source]

func_049  0x2acc  mdio_write(int phy, int reg, u16 v)

          [callers: eth_boot]

func_051  0x2ba4  eth_xmit(int hdr_len, const void *payload, int pay_len)
          one TX descriptor at 0xF07D0000: des0 = 0xF07D0080 (header buffer), des1 = payload, des2 = IOC | pay_len<<16 | hdr_len, des3 = OWN|FD|LD|CIC(3) | total; DMA ch0 TX tail 0xF02B1120 = 0xF07D0010; start TX (+0x1104 bit0) if DMA idle; wait +0x1160 bit0 (TI), W1C; wait OWN clear; stop TX; desc list +0x1114 = 0xF07D0000
          [callers: eth_boot]

func_053  0x2cb4  eth_boot()
          BS 4 or fallback. 'H'; CG+0 |= bit6; TOP+0[9:8]: 0 'E' internal PHY, 1 'O' RMII refclk out (PAD+0x98 |= bit10), 2 'I' RMII refclk in (TOP+0x14/0x18/0xe8 pinmux); wait CG+4 bit6; clocks CG+0x78 bit24, +0x74 bit29, reset CG+0x84 bit29; RMII: glue 0xF02B3004/0x3014 and external PHY reset on GPIO +0x3c/+0x5c/+0x7c bit6 (200/20/20/200 ms); internal: glue +4/+0xc and PHY analog bring-up 0xF02B38xx/0xF02B3Axx; DMA SWR (0xF02B1000 bit0) 1 s -> boot_fail(8); scan PHY addr 0..31 (reg1 != 0/0xffff); reg0 = 0x8000, wait reg1 bit5 4 s else "AN0"; reg9 &= ~0x300; print reg0 old/new; reg0 = (reg0 & ~0x2040) | 0x2200; wait "AN1" (timeout -> assume 100F); print reg5; 's' + "100F"/"100H"/"10F"/"10H" from reg5 bits 8-9/7/6/5 (none: K8); TRNG 0xF0680000+0xc -> MAC 00:00:00:xx:xx:xx and xid; MAC regs +0x300/+0x304, +0x90 = 1, +0x70 = 2, MAC_CFG = 0x08040003 (|0x2000 full duplex, |0xC000 100M / |0x8000 10M); 4 TX + 4 RX descriptors, RX buffers 0xF07C8000 + i*0x800; MTL +0xd30 = 0x23, +0xd00 = 2; 'X'; udelay 0.5 s; 0xF0FF0020 &= ~1; RX start. State machine (see eth state table): per frame 'R','I',"\r\n". On "Dn": memcpy32(0xF0800200, 0xF07C0000, 0x28), 0xF0FF0020 |= 1, check_and_run_loader(1, 0, 0)
          [callers: dispatch, next_boot_source]

          [callers: sd_boot]

func_060  0x439c  sd_boot()
          BS 1 or fallback. g_sd_xfer_mode = 0; 'S'; SDIO1 clock CG+0x74 bit2, +0x38 = 0x400, udelay 6; TOP+4/+0xa0 pinmux, PAD+0x68 = 0x1000000, CG+0x20 &= ~0x30, CG+0x3c div 0x1e9 (400 kHz, inferred); '1'; power/reset waits (+0 bit10, +0x4c bit0, 1 s -> K:); OTP bit11: SDIO+0x50 |= 0x1000 and 'F'; CMD0, CMD8 (0x1aa) -> g_sd_v2, ACMD41 loop (5 tries / 6000 polls, HCS if v2) -> g_sd_hc; CMD2, CMD3 -> g_sd_rca, CMD7, fast clock (CG+0x3c div 7, PAD+0x68 = 0x1111112), ACMD6 4-bit, CMD16 512; any failure -> boot_fail(2). Read MBR (K3), PBR (K4), scan up to 10 clusters from the first data sector for 0xAA55 at +0x30 (K5); extra header sectors (K8); hdr_script_run; copy header to load; read body (K7); check_and_run_loader(1, hdr+0x88>>31, SDIO1+0x6c + n(n-1)/2)
          [callers: dispatch, next_boot_source]


eMMC boot partition (BS 8/9)
